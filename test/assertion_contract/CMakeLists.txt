cmake_minimum_required(VERSION 3.21)
project(AssertionContract LANGUAGES C CXX)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT TARGET zeroerr)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." zeroerr EXCLUDE_FROM_ALL)
    target_compile_definitions(zeroerr PRIVATE ZEROERR_NO_MAIN)
    set(_contract_own_main ON)
endif()
foreach(mode IN ITEMS legacy strict)
    add_executable(assertion_contract_${mode} probe.cpp)
    target_link_libraries(assertion_contract_${mode} PRIVATE zeroerr)
    if(_contract_own_main)
        target_compile_definitions(assertion_contract_${mode} PRIVATE ZEROERR_ASSERTION_CONTRACT_MAIN)
    endif()
    if(mode STREQUAL "strict")
        target_compile_definitions(assertion_contract_${mode} PRIVATE ZEROERR_CHECK_AS_ERROR)
    endif()
endforeach()
add_test(NAME assertion_contract
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/verify.py"
            "$<TARGET_FILE:assertion_contract_strict>" "$<TARGET_FILE:assertion_contract_legacy>")
