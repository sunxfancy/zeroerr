cmake_minimum_required(VERSION 3.15)
project(ZeroErrCodeSize LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(USE_MOLD OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. zeroerr)
target_compile_definitions(zeroerr PUBLIC ZEROERR_NO_MAIN ZEROERR_DISABLE_AUTO_INIT)
add_executable(code_size_probe main.cpp)
target_link_libraries(code_size_probe PRIVATE zeroerr)
