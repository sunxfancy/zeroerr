cmake_minimum_required(VERSION 3.21)
project(NativeFixtureIdentity LANGUAGES C CXX)
include(CTest)
include(CheckIPOSupported)
check_ipo_supported(RESULT _ipo_supported)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ${_ipo_supported})
set(_fixture_sources first.cpp second.cpp)
if(NOT TARGET zeroerr)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." zeroerr EXCLUDE_FROM_ALL)
    target_compile_definitions(zeroerr PRIVATE ZEROERR_NO_MAIN)
    list(APPEND _fixture_sources main.cpp)
endif()
add_executable(fixture_identity ${_fixture_sources})
target_link_libraries(fixture_identity PRIVATE zeroerr)
add_test(NAME fixture_identity COMMAND fixture_identity --no-color)
