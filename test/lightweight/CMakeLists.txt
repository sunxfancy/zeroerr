cmake_minimum_required(VERSION 3.15)
project(ZeroErrLightweight LANGUAGES CXX)
if(NOT DEFINED CMAKE_CXX_STANDARD)
    set(CMAKE_CXX_STANDARD 17)
endif()
add_executable(string_test string_test.cpp ../../src/string.cpp)
target_include_directories(string_test PRIVATE ../../include)
add_library(core_headers OBJECT core_headers.cpp)
target_include_directories(core_headers PRIVATE ../../include)

file(GLOB_RECURSE amalgamation_inputs CONFIGURE_DEPENDS ../../include/*.h ../../src/*.cpp)
set(generator ${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/gen-single-file.cmake)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/zeroerr.hpp
    COMMAND ${CMAKE_COMMAND} -P ${generator}
    DEPENDS ${amalgamation_inputs} ${generator}
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
add_custom_target(lightweight_single_header DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/zeroerr.hpp)
add_executable(single_core single_implementation.cpp single_core.cpp include_order.cpp)
add_executable(single_optional single_optional.cpp)
foreach(target single_core single_optional)
    add_dependencies(${target} lightweight_single_header)
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
    endif()
endforeach()
enable_testing()
add_test(NAME InternalString COMMAND string_test)
add_test(NAME SingleHeaderCore COMMAND single_core --no-color)
add_test(NAME SingleHeaderXml COMMAND single_core --no-color --reporters=xml)
add_test(NAME SingleHeaderOptional COMMAND single_optional)
