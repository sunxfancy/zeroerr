138Stream& operator>>(Stream& stream, String& text) {
139 text.clear();
140 int c = stream.peek();
141 while (c != -1 && (c == ' ' || c == '\t' || c == '\n' || c == '\r')) {
142 stream.get();
143 c = stream.peek();
144 }
145 while (c != -1 && c != ' ' && c != '\t' && c != '\n' && c != '\r') {
146 text.push_back(static_cast<char>(stream.get()));
147 c = stream.peek();
148 }
149 if (text.empty()) stream.setstate(Stream::failbit);
150 return stream;
