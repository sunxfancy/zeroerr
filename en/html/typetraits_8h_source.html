289struct visit_impl {
290 template <typename T, typename F>
291 static void visit(T& tup, size_t idx, F&& fun) {
292 if (idx == I - 1)
293 fun(tuple_get<I - 1>(tup));

295 visit_impl<I - 1>::visit(tup, idx, std::forward<F>(fun));
296 }
