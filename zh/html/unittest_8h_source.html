339 template <typename T>
340 void operator()(T& arg) {
341 arg.reset();
342 for (size_t i = 0; i < arg.size(); ++i, ++arg) {
343 func();
344 }
345 }

348 void operator()(T& arg, Args&... args) {
349 arg.reset();

351 operator()(args...);
