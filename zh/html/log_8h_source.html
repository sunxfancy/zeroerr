523 // Low-level API: callers must externally synchronize metadata assignment
524 // and flush. Prefer LOG/ZEROERR_LOG for concurrent producers.
525 template <typename... T>
526 PushResult push(T&&... args) {
527 static_assert(alignof(LogMessageImpl<detail::to_store_type_t<T>...>) <= alignof(std::max_align_t),
528 "Over-aligned log arguments are not supported");
529 ZEROERR_LOCK(*mutex);
530 unsigned size = sizeof(LogMessageImpl<detail::to_store_type_t<T>...>);
531 void* p = alloc_block(size);
532 LogMessage* msg =
533 new (p) LogMessageImpl<detail::to_store_type_t<T>...>(std::forward<T>(args)...);
534 commit_block(ZEROERR_LOAD(m_last), size);
535 return {msg, size, *this};
536 }
537
538
539 // Construct arguments before acquiring the stream lock. Publish only complete
540 // records, so a concurrent flush cannot observe partially initialized data.
541 template <typename... T>
542 void write(LogMessageImpl<T...>&& message) {
543 static_assert(alignof(LogMessageImpl<T...>) <= alignof(std::max_align_t),
544 "Over-aligned log arguments are not supported");
545 ZEROERR_LOCK(*mutex);
546 void* storage = alloc_block(sizeof(message));
547 new (storage) LogMessageImpl<T...>(std::move(message));
548 auto* last = ZEROERR_LOAD(m_last);
549 commit_block(last, sizeof(message));
550 if (flush_mode == FLUSH_AT_ONCE) flush_unlocked();
551 }
