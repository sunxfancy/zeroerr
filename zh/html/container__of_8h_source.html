36struct ContainerOfBase {
37 int min_size = 0, max_size = 100, size = -1;
38
39 void WithMaxSize(unsigned value) { max_size = checkedSize(value); }
40 void WithMinSize(unsigned value) { min_size = checkedSize(value); }
41 void WithSize(unsigned value) { size = checkedSize(value); }
42
43protected:
44 static int checkedSize(unsigned value) {
45 if (value > static_cast<unsigned>((std::numeric_limits<int>::max)()))
46 throw std::invalid_argument("Container size exceeds supported range");
47 return static_cast<int>(value);
48 }
49
50 void validate() const {
51 if (size < -1 || min_size < 0 || max_size < min_size)
52 throw std::invalid_argument("Invalid container size bounds");
53 }
54
55 unsigned randomSize(Rng& rng) const {
56 validate();
57 if (size != -1) return static_cast<unsigned>(size);
58 return rng.bounded(static_cast<uint32_t>(max_size) -
59 static_cast<uint32_t>(min_size) + 1U) + min_size;
60 }
61
62 template <class Inner, class Corpus>
63 void mutate(Rng& rng, Corpus& v, bool only_shrink, const Inner& inner) const {
64 validate();
65 const size_t lower = static_cast<size_t>(size == -1 ? min_size : size);
66 const size_t upper = static_cast<size_t>(size == -1 ? max_size : size);
67 if (v.size() < lower || v.size() > upper)
68 throw std::invalid_argument("Corpus violates container size bounds");
69 const unsigned action = rng.bounded(5);
70 if (size == -1 && action == 0) {
71 if (v.size() > lower)
72 v.erase(v.begin() + rng.bounded(static_cast<uint32_t>(v.size())));
73 } else if (size == -1 && action == 1 && !only_shrink) {
74 if (v.size() < upper) v.push_back(inner.GetRandomCorpus(rng));
75 } else if (!v.empty()) {
76 inner.Mutate(rng, v[rng.bounded(static_cast<uint32_t>(v.size()))], only_shrink);
77 }
78 }

159// ContainerOf<T>(inner) combinator creates a domain for a container T (eg, a
160// std::vector, std::set, etc) where elements are created from `inner`.
161//
162// Example usage:
163//
164// ContainerOf<std::vector<int>>(InRange(1, 2021))
165//
166// The domain also supports customizing the minimum and maximum size via the
167// `WithSize`, `WithMinSize` and `WithMaxSize` functions. Eg:
168//
169// ContainerOf<std::vector<int>>(Arbitrary<int>()).WithMaxSize(5)
170//
171template <typename T, typename Inner>
172typename std::enable_if<detail::is_associative_container<T>::value,
173 AssociativeContainerOf<T, Inner>>::type
174ContainerOf(Inner&& inner) {
175 return AssociativeContainerOf<T, Inner>(std::move(inner));
176}
177
178template <typename T, typename Inner>
179typename std::enable_if<!detail::is_associative_container<T>::value &&
180 detail::is_container<T>::value,
181 SequenceContainerOf<T, Inner>>::type
182ContainerOf(Inner&& inner) {
183 return SequenceContainerOf<T, Inner>(std::move(inner));
184}
185
186// We can also support with a template template parameter, so that we can use
187// the name of the container instead of the complete type. In such case,
188// the ValueType of the inner domain should be passed into the container
189//
190// ContainerOf<std::vector>(Positive<int>()).WithSize(3);
191//
192template <template <typename, typename...> class T, typename... Inner,
193 typename C = T<typename Inner::ValueType...>>
194auto ContainerOf(Inner... inner) -> decltype(ContainerOf<C>(std::move(inner)...)) {
195 return ContainerOf<C>(std::move(inner)...);
196}
