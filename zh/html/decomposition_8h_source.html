131// Detect whether a type can be explicitly converted to bool. This covers
132// explicit operator bool (std::unique_ptr/std::shared_ptr), which
133// std::is_convertible would miss. Scoped enums are excluded so that a bare
134// CHECK(enum) does not silently turn into "value != 0"; unscoped enums keep
135// their historical implicit-conversion behavior.
136template <typename T, typename = void>
137struct has_bool_conversion : std::false_type {};
138
139template <typename T>
140struct has_bool_conversion<T, decltype(static_cast<bool>(std::declval<T>()), void())>
141 : std::integral_constant<
142 bool, !(std::is_enum<typename std::decay<T>::type>::value &&
143 !std::is_convertible<typename std::decay<T>::type, int>::value)> {};
144
145template <typename T>
146typename std::enable_if<has_bool_conversion<T>::value, bool>::type getBool(T&& lhs) {
147 return static_cast<bool>(lhs);
148}
149
150template <typename T>
151typename std::enable_if<!has_bool_conversion<T>::value, bool>::type getBool(T&&) {
152 return true;
153}
154} // namespace details

207struct ExpressionDecomposer {
208 // The right operator for capturing expressions is "<=" instead of "<<" (based on the
209 // operator precedence table) but then there will be warnings from GCC about "-Wparentheses"
210 // and since "_Pragma()" is problematic this will stay for now...
211 // https://github.com/catchorg/Catch2/issues/870
212 // https://github.com/catchorg/Catch2/issues/565
213
214 // For temporary objects, we need to use rvalue reference to avoid copy
215 template <typename L>
216 Expression<L> operator<<(L&& operand) {
217 return Expression<L>(static_cast<L&&>(operand));
218 }
219
220 // For other objects, we will store the reference
221 template <typename L,
222 typename std::enable_if<!std::is_rvalue_reference<L>::value, void>::type* = nullptr>
223 Expression<const L&> operator<<(const L& operand) {
224 return Expression<const L&>(operand);
225 }
226};

270public:
271 CombinedMatcher(IMatcherRef<T>&& lhs, IMatcherRef<T>&& rhs, bool is_and)
272 : lhs(std::move(lhs)), rhs(std::move(rhs)), is_and(is_and) {}
